cmake_minimum_required(VERSION 3.20)
project(raasta_api LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(httplib URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.3.tar.gz)
set(HTTPLIB_COMPILE OFF CACHE BOOL "Do not build cpp-httplib tests/examples" FORCE)
FetchContent_MakeAvailable(httplib)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(json)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MYSQL REQUIRED IMPORTED_TARGET mysqlclient)
add_executable(raasta-api server.cpp)
target_link_libraries(raasta-api PRIVATE httplib::httplib nlohmann_json::nlohmann_json PkgConfig::MYSQL)
target_compile_options(raasta-api PRIVATE -Wall -Wextra -Wpedantic)
